```python
import matplotlib.pyplot as plt
import numpy as np
from sklearn import datasets, linear_model
from sklearn.metrics import mean_squared_error, r2_score

# Load Dataset
diabetes = datasets.load_diabetes()

diabetes_X = diabetes.data[:, np.newaxis, 2]

# Examine one feature, that being, the second index element (third) column:
# 80 : 20 for training/testing - Split the data into training and testing
diabetes_X_train = diabetes_X[:-20] # All but last 20
diabetes_X_test = diabetes_X[-20:] # Last 20

# Split the targets into training and testing sets
diabetes_y_train = diabetes.target[:-20] # All but last 20
diabetes_y_test = diabetes.target[-20:]   # Last 20

# Linear Regression Module using Training Data
regr = linear_model.LinearRegression()

# Train the model using the training sets
regr.fit(diabetes_X_train, diabetes_y_train)

# Make predictions using the unseen testing set
diabetes_y_pred = regr.predict(diabetes_X_test)

# Statistics and Plot Outputs
print('Co-Efficients: ', regr_coef_)
print('Mean Square Error: %.2f' % mean_squared_error(diabetes_y_test, diabetes_y_pred))
print('Varience Score: %.2f' % r2_score(diabetes_y_test, diabetes_y_pred))

# Plot Outputs
plt.scatter(diabetes_X_test, diabetes_y_test, color="black")
plt.plot(diabetes_X_test, diabetes_y_pred, color="blue", linewidth=3)
plt.xticks(())
plt.yticks(())
plt.show()
```
